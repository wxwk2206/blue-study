## TCP介绍
```
通过Java程序来创建一个基于TCP协议的网络通讯程序

1.什么是TCP，它的优缺点是什么？
TCP--传输控制协议[有连接的协议]
传输数据
优点:保证数据安全可靠
缺点:消耗大，效率低

2.建立基于TCP的网络通讯程序需要使用的类和常用方法？
2.1  客户端程序
1.java.net.Socket类-- 创建基于TCP的通讯程序的客户端对象的java类
  通过这个类创建的对象就表示一个客户端
  构造方法
  Socket(InetAddress address, int port) 创建流套接字并将其连接到指定IP地址的指定端口号。 
  Socket(String host, int port) 创建流套接字并将其连接到指定主机上的指定端口号。
  实例方法
   OutputStream	getOutputStream() 返回此套接字的输出流。 
   InputStream	getInputStream() 返回此套接字的输入流。 
   void	close() 关闭此套接字。

2.服务器端程序
1.java.net.ServerSocket 创建基于TCP的通讯程序的服务器端对象的java类
  通过这个类创建的对象就表示一个服务器端
  构造方法
     ServerSocket(int port) 创建绑定到指定端口的服务器套接字。
  实例方法
  Socket	accept() 侦听要连接到此套接字并接受它。 【阻塞主线程运行】
   void	close() 关闭此套接字。 
```

## TCPClient
```java
package com.qiming.client;

import java.io.BufferedReader;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStream;
import java.net.InetAddress;
import java.net.Socket;
/**
 * 接收来客户端控制台上的数据，发送给服务器
 */
public class TCPClient {
    public static void main(String[] args) {
        //定义输入流来读取控制台数据
        BufferedReader conreader = null;
        //定义保存控制台数据的字符串变量；
        String info = null;
        //定义保存客户端对象的变量
        Socket client = null;
        //定义服务器的ip
        InetAddress serverip = null;
        //String  serveripString = null;
        //定义服务器端口
        int  serverpost = 8000;
        //定义客户端写出数据的输出流对象的变量
        OutputStream clientOut = null;
        //定义一个boolean值来控制客户端的持续发送
        boolean clientflag = true;
        //定义读取服务器端返回的数据的输入流对象变量
        InputStream clientRead = null;
        try {
            //创建客户端对象
            serverip = InetAddress.getLocalHost();
            //serveripString = "127.0.0.1";
            client = new Socket(serverip, serverpost);
            System.out.println("客户端连接服务成功......");
            //打开菜单
            ClienrtHelper.clientMnue();
            conreader = new BufferedReader(new InputStreamReader(System.in));
            clientOut = client.getOutputStream();
            //创建客户端读取数据的输入流对象
            clientRead = client.getInputStream();
            while(clientflag) {
                //发送
                System.out.println("请输入指定的命令，完成对应的功能:");
                info = conreader.readLine();
                System.out.println("来自客户端控制台的数据===" + info);
                clientOut.write(info.getBytes());
                if(info.equals("886")){
                    clientflag = false;
                    System.out.println("退出服务器");
                }else {
                    //接收
                    //读取服务器端返回来的数据
                    byte dadabyte[] = new byte[100];
                    int len = clientRead.read(dadabyte);
                    //将保存有接收数据的字节数组转换称String
                    String msg = new String(dadabyte, 0, len);
                    System.out.println("来自服务器端的数据===" + msg);
                }
            }
            //关闭
            clientOut.close();
            client.close();
            conreader.close();
        }catch (Exception e){
            e.printStackTrace();
        }
    }
}
```

## TCPServer
```java
package com.qiming.server;

import java.io.InputStream;
import java.io.OutputStream;
import java.net.ServerSocket;
import java.net.Socket;

/**
 * 接收客户端发送来的字符串数据
 */
public class TCPServer {
    public static void main(String[] args) {
        //定义服务器开发的端口号
        int serverport = 8000;
        //定义保存服务器的变量
        ServerSocket   serverSocket = null;

        boolean  flag = true;
        try {
            //创建服务器对象
            serverSocket = new ServerSocket(serverport);
            System.out.println("服务器启动，等待客户端进入.......");
            //持续接收客户端
            while(flag){
                try {
                    //监听服务器的开发端口，等待客户端连接
                    Socket clientSocket = serverSocket.accept();
                    System.out.println("客户端已经连接，准备收发数据......");
                    //启动客户端线程独立运行
                    new ClientThread(clientSocket).start();
                }catch (Exception e){
                    e.printStackTrace();
                    flag=false;
                }
            }
            serverSocket.close();
        }catch (Exception e){
            e.printStackTrace();
        }
    }
}
```

## ClientThread
```java
package com.qiming.server;

import java.io.InputStream;
import java.io.OutputStream;
import java.net.Socket;

/**
 * 连接到服务器上的每一个客户端的对立操作线程
 */
public class ClientThread extends Thread{
    //定义保存连接到服务器的客户端对象的变量
    private  Socket clientSocket = null;
    //定义boolean值控制服务器持续接收数据
    private boolean  serverflag = true;
    //定义读取客户端数据的输入流对象变量
    private InputStream serverRead = null;
    //定义服务器写出数据的输出流对象的变量
    private OutputStream serverOut = null;
    // ⭐ 构造函数：接收 TCPServer accept() 建立好的连接
    public ClientThread(Socket socket){
        clientSocket = socket;
    }
    /**
     * 线程操作方法
     */
    @Override
    public void run() {
        try {
            serverRead = clientSocket.getInputStream();
            serverOut = clientSocket.getOutputStream();
            while (serverflag) {
                //读取客户端发送来的数据
                byte dadabyte[] = new byte[100];
                int len = serverRead.read(dadabyte);
                //将保存有接收数据的字节数组转换称String
                String msg = new String(dadabyte, 0, len);
                System.out.println("来自客户端的数据===" + msg);
                if (msg.equals("886")) {
                    serverflag = false;
                } else {
                    //发送数据
                    msg = "server-" + msg;
                    serverOut.write(msg.getBytes());
                }
            }
            serverOut.close();
            serverRead.close();
            clientSocket.close();
        }catch(Exception e){
            e.printStackTrace();
        }
    }
}
```
