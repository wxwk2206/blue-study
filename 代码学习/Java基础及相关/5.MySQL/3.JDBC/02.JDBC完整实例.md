### JDBC访问数据库需要用到的类、接口、方法
 1、java.sql包DriverManager类---
负责加载各种不同驱动程序（Driver），并根据不同的请求，向调用者返回相应的数据库连接。
1.1 数据驱动程序（Driver）--- 根据所连接的数据库不同需要下载对应数据库的驱动程序。
```xml
<!-- 导入Mysql数据库链接jar包 -->
    <dependency>
      <groupId>mysql</groupId>
      <artifactId>mysql-connector-java</artifactId>
      <version>8.0.33</version>
    </dependency>
```

1.2 向调用者返回相应的数据库连接-----DriverManager类中的静态方法getConnection(url,username,password)向调用者返回相应的数据库连接


2.java.sql包Connection接口---数据库连接，负责与进行数据库间通讯，SQL执行以及事务处理都是在某个特定Connection环境中进行的。可以产生用以执行SQL的Statement。
2.1产生用以执行SQL的Statement。
    \[用以执行SQL查询和更新（针对静态SQL语句和单次执行）] 
    Statement  createStatement()

2.2产生用以执行SQL的PreparedStatement。
        \[用以执行包含动态参数的SQL查询和更新（在服务器端编译，允许重复执行以提高效率）]
       PreparedStatement  prepartatement(Sql)
       参数sql---包含动态参数的SQL语句
       select   * from  t_user  where uid=？;【需要给sql动态传递参数值】
       ？---占位符


2.3 java.sql包Statement接口-----静态SQL语句执行【不安全】
      int executeUpdate(SQL)----以执行SQL更新【insert / update /delete】
      ResultSet    executeQuery(SQL)----以执行SQL查询【select】


2.4 java.sql包PreparedStatement接口--用以执行包含动态参数的SQL查询和更新（在服务器端编译，允许重复执行以提高效率）
          int executeUpdate()---- 以执行SQL更新【insert / update /delete】
          ResultSet    executeQuery()----以执行SQL查询【select】
         动态参数的SQL语句，需要给sql动态传递参数值、
         例如:   select   * from  t_user  where uid=？;
                    ？---占位符
         有一组填补sql语句占位符的方法
         setInt(参数1，参数2)  int--int
         setLong(参数1，参数2)  long---bigint
         setDouble(参数1，参数2) double--double       
         setString(参数1，参数2) String--String      
         ........

2.5.java.sql包ResultSet接口 --- 数据库结果集的数据表，通常通过执行查询数据库的语句生成。
	ResultSet 对象具有指向其当前数据行的指针。最初，指针被置于第一行之前。next 方法将指针移动到下一行；因为该方法在 ResultSet 对象中没有下一行时返回 false，所以可以在 while 循环中使用它来迭代结果集。
	默认的 ResultSet 对象不可更新，仅有一个向前移动的指针。因此，只能迭代它一次，并且只能按从第一行到最后一行的顺序进行。
ResultSet会有一组获取指定列数据的方法。
getXXXX(int colindex)---int colindex\[当前需要获取数据值的列在当前行中的位置【从0开始】]
getXXXX(String colname)--String colname\[当前需要获取数据值的列名称]

```java
package com.qiming.dbconn;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;

/**
 * 数据库连接操作类
 * 1、加载数据库驱动
 * 2.得到数据库连接对象
 */
public class DbConnection {
    //定义保存数据库驱动名称的静态成员变量
    private final static String  DRIVERNAME = "com.mysql.cj.jdbc.Driver";
    //定义保存数据库位置的的静态成员变量
    private final static String   URL= "jdbc:mysql://127.0.0.1:3306/qiming_db";
    //定义保存登录数据库的用户名的静态成员变量
    private final static String   USERNAME= "root";
    //定义保存登录数据库位置的密码的静态成员变量
    private final static String   PASSWORD= "123456";
    //通过Java中的静态代码块,加载数据库驱动程序
    //数据库驱动程序----是我们导入的驱动包中的一个Java类
    //驱动包中的驱动类的全名称：com.mysql.cj.jdbc.Driver  【适用于MySQL8.0以上】
    //驱动包中的驱动类的全名称： com.mysql.jdbc.Driver  【适用于MySQL5.5  5.6  5.7 】
    /*
      public class com.mysql.jdbc.Driver extends com.mysql.cj.jdbc.Driver {
     */
    static{
        try {
            //通过Java的反射机制来加载数据库驱动
            Class.forName(DRIVERNAME);
        }catch(ClassNotFoundException e){
            System.out.println("数据库驱动加载失败");
        }
    }

    //创建一个静态方法，来得到数据连接对象
    public  static Connection  getConnection(){
        //定义保存数据库连接对象的变量
        Connection connection=null;
        //当数据库连接对象为空时创建数据库连接对象
        if(connection == null){
            try {
                //DriverManager类中的静态方法getConnection()需要3个参数
                //参数1---url---需要连接的数据库所在位置
                //url----数据库访问方式【jdbc】:数据库类型名称【mysql】://数据库所在计算机的ip地址:端口【3306】/数据库名【qiming_db】
                //jdbc:mysql://127.0.0.1:3306/qiming_db
                //参数2---username---登录数据库的账户名【root】
                //参数3---password---登录数据库的密码
                connection = DriverManager.getConnection(URL, USERNAME, PASSWORD);
            }catch (SQLException e){
                e.printStackTrace();
                System.out.println("数据库连接获取失败！");
            }
            }
        return  connection;
    }
}
```

## 6.2dao层PreparedStatement接口
1.2  PreparedStatement接口--用以执行包含动态参数的SQL查询和更新
```java
package com.qiming.dao;

import com.qiming.dbconn.DbConnection;
import com.qiming.javabean.UserBean;

import java.sql.*;
import java.util.ArrayList;
import java.util.List;

public class UserBeanDao {
    /**
     * 添加一个用户信息的方法
     * @param userBean
     * @return
     */
    public boolean insertUserBean(UserBean userBean){
        //定义保存返回值结果的boolean类型的变量
        boolean  insertflag = false;
        //定义保存数据库连接对象的变量
        Connection connection = null;
        //定义保存PreparedStatement对象的变量
        PreparedStatement preparedStatement = null;
        try {
            //得到数据连接对象Connection
            connection = DbConnection.getConnection();
            //添加数据的sql语句
            String insertsql = "insert  into t_user values(null,?,?,?);";
            //通过Connection对象创建出PreparedStatement对象
            preparedStatement = connection.prepareStatement(insertsql);
            //为？占位符补值
            preparedStatement.setString(1,userBean.getUsername());
            preparedStatement.setInt(2,userBean.getUserage());
            preparedStatement.setString(3,userBean.getUseraddress());
            //执行上面的insertsql语句
            int  inserttemp = preparedStatement.executeUpdate();
            //判断添加成功函数失败
            if(inserttemp > 0){
                insertflag = true;
            }else{
                insertflag = false;
            }
```
